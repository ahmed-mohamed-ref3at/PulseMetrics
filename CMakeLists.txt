cmake_minimum_required(VERSION 3.20)
project(PulseMetrics VERSION 2.0 LANGUAGES C)
if(NOT WIN32)
  message(FATAL_ERROR "PulseMetrics requires Windows and the UCRT64 toolchain")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
add_library(pulse_sqlite STATIC native/vendor/sqlite3.c)
target_include_directories(pulse_sqlite PUBLIC native/vendor)
target_compile_definitions(pulse_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION=1)
add_library(pulse_json STATIC native/vendor/cJSON.c)
target_include_directories(pulse_json PUBLIC native/vendor)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS native/core/src/*.c)
add_library(pulsecore STATIC ${CORE_SOURCES})
target_include_directories(pulsecore PUBLIC native/core/include)
target_link_libraries(pulsecore PUBLIC pulse_sqlite pulse_json ws2_32)
target_compile_definitions(pulsecore PRIVATE _CRT_SECURE_NO_WARNINGS)
add_library(pulsedata STATIC native/data_access/src/data_access.c)
target_include_directories(pulsedata PUBLIC native/data_access/include)
target_link_libraries(pulsedata PUBLIC pulse_sqlite)
foreach(component collector monitor)
  add_executable(pulse${component} native/${component}/src/main.c native/${component}/src/${component}.c)
  target_include_directories(pulse${component} PRIVATE native/${component}/include)
  target_link_libraries(pulse${component} PRIVATE pulsecore)
endforeach()
if(BUILD_TESTING)
  add_executable(test_contract native/tests/test_contract.c)
  target_link_libraries(test_contract PRIVATE pulsecore)
  target_compile_options(test_contract PRIVATE -UNDEBUG -Wall -Wextra)
  add_test(NAME v2_contract COMMAND test_contract)
  foreach(name parser metrics aggregator data_access integration)
    add_executable(test_${name} native/tests/test_${name}.c)
    target_link_libraries(test_${name} PRIVATE pulsecore pulsedata)
    target_include_directories(test_${name} PRIVATE native/collector/include)
    if(name STREQUAL integration)
      target_sources(test_${name} PRIVATE native/collector/src/collector.c)
    endif()
    target_compile_options(test_${name} PRIVATE -UNDEBUG)
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test-run/${name}")
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/test-run/${name}" TIMEOUT 30)
  endforeach()
endif()
